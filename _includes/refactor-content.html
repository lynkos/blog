{% assign WIDGET_TAGS = 'p,dd,td' | split: ',' %}

{% for _widget_tag in WIDGET_TAGS %}
  {% assign _open_marker = '<' | append: _widget_tag %}
  {% assign _close_tag = '' %}

  {% unless _content contains _close_tag %}
    {% continue %}
  {% endunless %}

  {% assign _element_snippets = _content | split: _close_tag %}
  {% assign _new_content = '' %}

  {% for _snippet in _element_snippets %}
    {% if forloop.last %}
      {% assign _new_content = _new_content | append: _snippet %}
      {% continue %}
    {% endif %}

    
    {% assign _element_fragment = _snippet
      | split: _open_marker
      | last
    %}

    {% assign _fragment_size = _element_fragment | size %}
    {% assign _snippet_size = _snippet | size %}
    {% assign _marker_size = _open_marker | size %}

    {% assign _prefix_size = _snippet_size
      | minus: _fragment_size
      | minus: _marker_size
    %}

    {% assign _prefix = _snippet | slice: 0, _prefix_size %}

    
    {% assign _pre_open_count = _prefix | split: '' | size %}
    {% assign _code_open_count = _prefix | split: '' | size %}

    {% assign _inside_code = false %}

    {% if _pre_open_count > _pre_close_count %}
      {% assign _inside_code = true %}
    {% endif %}

    {% if _code_open_count > _code_close_count %}
      {% assign _inside_code = true %}
    {% endif %}

    {% if _inside_code %}
      {% assign _new_content = _new_content
        | append: _snippet
        | append: _close_tag
      %}
      {% continue %}
    {% endif %}

    
    {% assign _opening_fragment = _element_fragment
      | split: '>'
      | first
    %}

    {% assign _element_content = _element_fragment
      | remove_first: _opening_fragment
      | remove_first: '>'
    %}

    {% assign _clean_opening_fragment = _opening_fragment %}
    {% assign _widget_type = '' %}
    {% assign _widget_value = '' %}

    
    {% if _opening_fragment contains ' rating="' %}
      {% assign _widget_type = 'rating' %}

      {% assign _widget_value = _opening_fragment
        | split: ' rating="'
        | last
        | split: '"'
        | first
      %}

      {% assign _widget_attribute = ' rating="'
        | append: _widget_value
        | append: '"'
      %}

      {% assign _clean_opening_fragment = _clean_opening_fragment
        | remove_first: _widget_attribute
      %}

    {% elsif _opening_fragment contains ' price="' %}
      {% assign _widget_type = 'price' %}

      {% assign _widget_value = _opening_fragment
        | split: ' price="'
        | last
        | split: '"'
        | first
      %}

      {% assign _widget_attribute = ' price="'
        | append: _widget_value
        | append: '"'
      %}

      {% assign _clean_opening_fragment = _clean_opening_fragment
        | remove_first: _widget_attribute
      %}
    {% endif %}

    
    {% if _widget_type == '' and _widget_tag == 'td' %}
      {% assign _normalized_content = _element_content
        | strip
        | replace: '“', '"'
        | replace: '”', '"'
      %}

      {% if _normalized_content contains '{: rating="' %}
        {% assign _candidate_value = _normalized_content
          | split: '{: rating="'
          | last
          | split: '"'
          | first
        %}

        {% assign _expected_literal = '{: rating="'
          | append: _candidate_value
          | append: '" }'
        %}

        
        {% if _normalized_content == _expected_literal %}
          {% assign _widget_type = 'rating' %}
          {% assign _widget_value = _candidate_value %}
          {% assign _element_content = '' %}
        {% endif %}

      {% elsif _normalized_content contains '{: price="' %}
        {% assign _candidate_value = _normalized_content
          | split: '{: price="'
          | last
          | split: '"'
          | first
        %}

        {% assign _expected_literal = '{: price="'
          | append: _candidate_value
          | append: '" }'
        %}

        
        {% if _normalized_content == _expected_literal %}
          {% assign _widget_type = 'price' %}
          {% assign _widget_value = _candidate_value %}
          {% assign _element_content = '' %}
        {% endif %}
      {% endif %}
    {% endif %}

    {% assign _widget_html = '' %}

    
    {% if _widget_type == 'rating' %}
      {% assign _rating = _widget_value
        | times: 1.0
        | at_least: 0
        | at_most: 5
      %}

      {% assign _full = _rating | floor %}
      {% assign _remainder = _rating | minus: _full %}
      {% assign _half = 0 %}

      
      {% if _remainder >= 0.75 %}
        {% assign _full = _full | plus: 1 %}
      {% elsif _remainder >= 0.25 %}
        {% assign _half = 1 %}
      {% endif %}

      {% assign _half_index = _full | plus: 1 %}
      {% assign _icons = '' %}

      {% for i in (1..5) %}
        {% if i <= _full %}
          {% assign _icons = _icons
            | append: ''
          %}
        {% elsif _half == 1 and i == _half_index %}
          {% assign _icons = _icons
            | append: ''
          %}
        {% else %}
          {% assign _icons = _icons
            | append: ''
          %}
        {% endif %}
      {% endfor %}

      {% assign _accessible_text = _rating
        | append: ' out of 5 stars'
      %}

      {% assign _widget_html = ''
        | append: _accessible_text
        | append: ''
        | append: ''
        | append: _icons
        | append: ''
      %}

    
    {% elsif _widget_type == 'price' %}
      {% assign _price = _widget_value
        | times: 1
        | at_least: 0
        | at_most: 5
      %}

      {% assign _icons = '' %}

      {% if _price > 0 %}
        {% for i in (1.._price) %}
          {% assign _icons = _icons
            | append: ''
          %}
        {% endfor %}
      {% endif %}

      {% assign _accessible_text = _price | append: ' dollar sign' %}

      {% unless _price == 1 %}
        {% assign _accessible_text = _accessible_text | append: 's' %}
      {% endunless %}

      {% assign _widget_html = ''
        | append: _accessible_text
        | append: ''
        | append: ''
        | append: _icons
        | append: ''
      %}
    {% endif %}

    
    {% assign _new_content = _new_content
      | append: _prefix
      | append: _open_marker
      | append: _clean_opening_fragment
      | append: '>'
      | append: _element_content
      | append: _widget_html
      | append: _close_tag
    %}
  {% endfor %}

  {% assign _content = _new_content %}
{% endfor %}



{% assign IMG_TAG = '' | first %}
    {% assign _right = _img_snippet | remove: _left %}

    {% unless _left contains 'src=' %}
      {% continue %}
    {% endunless %}

    {% assign _left = _left | remove: ' /' | replace: ' w=', ' width=' | replace: ' h=', ' height=' %}
    {% assign _attrs = _left | split: '" ' %}

    {% assign _src = null %}
    {% assign _lqip = null %}
    {% assign _class = null %}

    {% for _attr in _attrs %}
      {% unless _attr contains '=' %}
        {% continue %}
      {% endunless %}

      {% assign _pair = _attr | split: '="' %}
      {% capture _key %}{{ _pair | first }}{% endcapture %}
      {% capture _value %}{{ _pair | last | remove: '"' }}{% endcapture %}

      {% case _key %}
        {% when 'src' %}
          {% assign _src = _value %}
        {% when 'lqip' %}
          {% assign _lqip = _value %}
        {% when 'class' %}
          {% assign _class = _value %}
      {% endcase %}
    {% endfor %}

    
    {% assign _wrapper_class = '' %}

    {% if _class %}
      {% assign _remain = _class %}
      {% assign _class_array = _class | split: ' ' %}

      {% for c in _class_array %}
        {% assign _pick = false %}

        {% case c %}
          {% when 'preview-img', 'normal', 'left', 'right', 'light', 'dark' %}
            {% assign _pick = true %}
          {% else %}
            {% if c contains '-' %}
              {% assign start = c | split: '-' | first %}
              {% if start == 'w' %}
                {% assign _pick = true %}
              {% endif %}
            {% endif %}
        {% endcase %}

        {% if _pick %}
          {% assign _remain = _remain | remove: c | strip %}
          {% assign _wrapper_class = _wrapper_class | append: ' ' | append: c %}
        {% endif %}
      {% endfor %}

      {% unless _wrapper_class == '' %}
        {% capture _old_class %}class="{{ _class }}"{% endcapture %}
        {% assign _left = _left | remove: _old_class %}
        {% unless _remain == '' %}
          {% capture _new_class %}class="{{ _remain }}"{% endcapture %}
          {% assign _left = _left | append: _new_class %}
        {% endunless %}
      {% endunless %}
    {% endif %}

    {% assign _final_src = null %}
    {% assign _lazyload = true %}

    {%- capture _img_url -%}
      {% include media-url.html src=_src subpath=page.media_subpath %}
    {%- endcapture -%}

    {% assign _path_prefix = _img_url | remove: _src %}

    {% unless _src contains '//' %}
      {% assign _final_src = _path_prefix | append: _src %}
      {% assign _src_alt = 'src="' | append: _path_prefix %}
      {% assign _left = _left | replace: 'src="', _src_alt %}
    {% endunless %}

    {% if _lqip %}
      {% assign _lazyload = false %}
      {% assign _wrapper_class = _wrapper_class | append: ' blur' %}

      {% unless _lqip contains 'data:' %}
        {% assign _lqip_alt = 'lqip="' | append: _path_prefix %}
        {% assign _left = _left | replace: 'lqip="', _lqip_alt %}
      {% endunless %}

      
      {% assign _left = _left | replace: 'src=', 'data-src=' | replace: ' lqip=', ' data-lqip="true" src=' %}

    {% else %}
      {% assign _wrapper_class = _wrapper_class | append: ' shimmer' %}
    {% endif %}

    
    {% if _lazyload %}
      {% assign _left = _left | append: ' loading="lazy"' %}
    {% endif %}

    {% if page.layout == 'home' %}
      
      {% assign _wrapper_start = '
' %}

      {% assign _img_content = _img_content | append: _wrapper_start %}
      {% assign _right = _right | prepend: '>` is wrapped by `` -->
      {% assign _parent = _right | slice: 1, 4 %}

      {% if _parent == '' %}
        
        {% assign _size = _img_content | size | minus: 1 %}
        {% capture _class %}
          class="img-link{% unless _lqip %} shimmer{% endunless %}"
        {% endcapture %}
        {% assign _img_content = _img_content | slice: 0, _size | append: _class | append: '>' %}

      {% else %}
        
        {% assign _wrapper_start = _final_src
          | default: _src
          | prepend: '' %}

        {% assign _img_content = _img_content | append: _wrapper_start %}
        {% assign _right = '>
    {% assign _img_content = _img_content | append: IMG_TAG | append: _left | append: _right %}
  {% endfor %}

  {% if _img_content %}
    {% assign _content = _img_content %}
  {% endif %}
{% endif %}
